const CANDLE_HISTORY_PAGE_SIZE = 1500;
const CANDLE_HISTORY_PREFETCH_PAGES = 2;
const CANDLE_CACHE_DB = 'binance-micro-terminal-cache';
const CANDLE_CACHE_STORE = 'histories';
const CANDLE_CACHE_MAX_CANDLES = 100000;
const candleCacheWriteTimers = new Map();
let candleCacheDbPromise = null;

// IndexedDB keeps large candle histories without blocking the UI or exhausting localStorage.
function openCandleCache() {
    if (!('indexedDB' in window)) return Promise.resolve(null);
    if (candleCacheDbPromise) return candleCacheDbPromise;
    candleCacheDbPromise = new Promise(resolve => {
        const request = indexedDB.open(CANDLE_CACHE_DB, 1);
        request.onupgradeneeded = () => {
            const db = request.result;
            if (!db.objectStoreNames.contains(CANDLE_CACHE_STORE)) db.createObjectStore(CANDLE_CACHE_STORE, { keyPath: 'key' });
        };
        request.onsuccess = () => resolve(request.result);
        request.onerror = () => { console.warn('[Cache] IndexedDB unavailable:', request.error); resolve(null); };
        request.onblocked = () => resolve(null);
    });
    return candleCacheDbPromise;
}

async function readCandleCache(symbol, timeframe) {
    const db = await openCandleCache();
    if (!db) return null;
    return new Promise(resolve => {
        try {
            const request = db.transaction(CANDLE_CACHE_STORE, 'readonly').objectStore(CANDLE_CACHE_STORE).get(`${symbol}|${timeframe}`);
            request.onsuccess = () => {
                const record = request.result;
                const candles = Array.isArray(record?.candles) ? record.candles.map(normalizeServiceCandle).filter(Boolean) : [];
                resolve(candles.length ? { candles, savedAt: record.savedAt || 0 } : null);
            };
            request.onerror = () => resolve(null);
        } catch (_) { resolve(null); }
    });
}

async function writeCandleCache(symbol, timeframe, sourceCandles) {
    const db = await openCandleCache();
    if (!db || !sourceCandles?.length) return;
    const candles = sourceCandles.slice(-CANDLE_CACHE_MAX_CANDLES);
    return new Promise(resolve => {
        try {
            const tx = db.transaction(CANDLE_CACHE_STORE, 'readwrite');
            tx.objectStore(CANDLE_CACHE_STORE).put({ key: `${symbol}|${timeframe}`, savedAt: Date.now(), candles });
            tx.oncomplete = () => resolve(true);
            tx.onerror = tx.onabort = () => resolve(false);
        } catch (_) { resolve(false); }
    });
}

function scheduleCandleCacheSave(tab) {
    if (!tab?.rawCandles?.length) return;
    const symbol = tab.symbol;
    const timeframe = tab.timeframe;
    const snapshot = tab.rawCandles.slice(-CANDLE_CACHE_MAX_CANDLES);
    const key = `${symbol}|${timeframe}`;
    const oldTimer = candleCacheWriteTimers.get(key);
    if (oldTimer) clearTimeout(oldTimer);
    candleCacheWriteTimers.set(key, setTimeout(() => {
        candleCacheWriteTimers.delete(key);
        void writeCandleCache(symbol, timeframe, snapshot);
    }, 1200));
}

function normalizeServiceCandle(c) {
    if (!c) return null;
    let time = Number(c.time);
    if (!Number.isFinite(time)) {
        const raw = c.open_time ?? c.openTime;
        time = typeof raw === 'number' ? raw : Date.parse(raw);
    }
    if (!Number.isFinite(time)) return null;
    if (time > 1e12) time = Math.floor(time / 1000);
    else time = Math.floor(time);
    const open = Number(c.open), high = Number(c.high), low = Number(c.low), close = Number(c.close);
    const volume = Number(c.volume ?? 0);
    if (![open, high, low, close].every(Number.isFinite)) return null;
    return { time, open, high, low, close, volume: Number.isFinite(volume) ? volume : 0 };
}

function sendTabSubscription(tab) {
    if (!tab || !tab.registered || !wsConn || wsConn.readyState !== WebSocket.OPEN) return;
    const key = `${tab.symbol}:${tab.timeframe}`;
    if (wsSubscriptions.has(key)) return;
    wsConn.send(JSON.stringify({ symbol: tab.symbol, timeframe: tab.timeframe }));
    wsSubscriptions.add(key);
}

function subscribeTabToService(tab) {
    if (wsConn && wsConn.readyState === WebSocket.OPEN) sendTabSubscription(tab);
}

// The service protocol has subscribe but no unsubscribe message.
// Reconnect on timeframe changes to clear stale server-side subscriptions safely.
function resetWebSocketSubscriptions() {
    wsSubscriptions = new Set();
    if (wsConn && (wsConn.readyState === WebSocket.OPEN || wsConn.readyState === WebSocket.CONNECTING)) {
        try { wsConn.close(1000, 'timeframe changed'); }
        catch (e) { console.warn('[Candle Service] WebSocket reset failed', e); }
    } else {
        initWebSocket();
    }
}

function sendAllTabSubscriptions() {
    appState.tabs.forEach(sendTabSubscription);
}

function initWebSocket() {
    if (wsConn && (wsConn.readyState === WebSocket.OPEN || wsConn.readyState === WebSocket.CONNECTING)) return;
    if (wsReconnectTimer) { clearTimeout(wsReconnectTimer); wsReconnectTimer = null; }
    try {
        wsConn = new WebSocket(CANDLE_SERVICE_WS_URL);
    } catch (e) {
        console.error('[Candle Service] WebSocket initialization failed', e);
        scheduleWebSocketReconnect();
        return;
    }

    wsConn.onopen = () => {
        console.info('[Candle Service] Client WebSocket connected');
        wsReconnectDelay = 1000;
        wsSubscriptions = new Set();
        sendAllTabSubscriptions();
    };

    wsConn.onmessage = (ev) => {
        try {
            const msg = JSON.parse(ev.data);
            if (msg.type === 'CONNECTED') { sendAllTabSubscriptions(); return; }
            if (msg.type === 'SUBSCRIBED') {
                console.info(`[Candle Service] Subscribed ${msg.symbol} ${msg.timeframe}`);
                return;
            }
            if (msg.type === 'ERROR') {
                console.warn('[Candle Service] WebSocket error:', msg.message);
                return;
            }
            if (msg.type === 'HISTORY_BURST') return; // HTTP paging is authoritative; WS is used for live updates.
            if (!['CANDLE_UPDATE', 'CANDLE_UPDATED'].includes(msg.type) || !msg.symbol || !msg.candle) return;

            const tab = appState.tabs.find(t => t.symbol === msg.symbol && t.timeframe === msg.timeframe);
            if (!tab) return;
            const normalized = normalizeServiceCandle(msg.candle);
            if (!normalized) return;
            applyServerCandle(tab, normalized);
            checkTabTPSL(tab, normalized.close);
            if (appState.activeTabId === tab.id) updateActiveTabUI();
            updateGlobalStats();
        } catch (e) {
            console.error('[Candle Service] Invalid WebSocket message', e);
        }
    };

    wsConn.onerror = (e) => console.warn('[Candle Service] WebSocket connection error', e);
    wsConn.onclose = () => {
        wsSubscriptions = new Set();
        scheduleWebSocketReconnect();
    };
}

function scheduleWebSocketReconnect() {
    if (wsReconnectTimer) return;
    const wait = wsReconnectDelay;
    wsReconnectDelay = Math.min(wsReconnectDelay * 2, 15000);
    wsReconnectTimer = setTimeout(() => {
        wsReconnectTimer = null;
        initWebSocket();
    }, wait);
}
