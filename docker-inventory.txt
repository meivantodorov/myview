----- FILE: backend\scripts\backup.bat -----
@echo off
cd /d "%~dp0.."
if not exist backups mkdir backups
for /f %%i in ('powershell -NoProfile -Command "Get-Date -Format yyyyMMdd-HHmmss"') do set STAMP=%%i
docker compose exec -T db sh -c "pg_dump -U \"$POSTGRES_USER\" \"$POSTGRES_DB\"" > "backups\\candles-%STAMP%.sql"
if errorlevel 1 echo Backup failed. Check database container health and permissions.


----- FILE: backend\scripts\doctor.bat -----
@echo off
cd /d "%~dp0.."
echo === Docker version ===
docker --version
echo === Compose version ===
docker compose version
echo === Containers ===
docker compose ps
echo === API readiness ===
powershell -NoProfile -Command "try { Invoke-RestMethod http://localhost:8000/health/ready | ConvertTo-Json -Depth 5 } catch { Write-Host $_; exit 1 }"
echo === Recent API logs ===
docker compose logs --tail=80 api


----- FILE: backend\scripts\logs.bat -----
@echo off
cd /d "%~dp0.."
docker compose logs -f --tail=200 %*


----- FILE: backend\scripts\start.bat -----
@echo off
setlocal
cd /d "%~dp0.."
if not exist .env (
  copy .env.example .env >nul
  echo Created .env from .env.example. Review POSTGRES_PASSWORD before exposing this service.
)
docker compose up --build -d
if errorlevel 1 (
  echo Docker Compose failed. Check Docker Desktop and run scripts\doctor.bat
  exit /b 1
)
echo Service starting. API: http://localhost:8000  Docs: http://localhost:8000/docs
echo Logs: docker compose logs -f api


----- FILE: backend\scripts\start.ps1 -----
$ErrorActionPreference = "Stop"
Set-Location (Join-Path $PSScriptRoot "..")
if (!(Test-Path ".env")) { Copy-Item ".env.example" ".env"; Write-Host "Created .env; review its credentials." }
docker compose up --build -d
if ($LASTEXITCODE -ne 0) { throw "Docker Compose failed. Check Docker Desktop." }
Write-Host "API: http://localhost:8000 | Swagger: http://localhost:8000/docs"


----- FILE: backend\scripts\stop.bat -----
@echo off
cd /d "%~dp0.."
docker compose down


----- FILE: backend\scripts\test.bat -----
@echo off
setlocal
cd /d "%~dp0.."
echo Running unit tests in a temporary container...
docker compose run --rm --no-deps --entrypoint sh api -c "pip install --no-cache-dir pytest pytest-asyncio && pytest -q"
if errorlevel 1 (
  echo Tests failed. Review the output above.
  exit /b 1
)
echo All tests passed.
